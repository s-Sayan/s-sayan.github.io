	\section{Research Experience}
	
	\cvitem{Postdoctoral Research}{Developing the end-to-end image processing and weak-lensing pipeline for SuperBIT, NASA's balloon-borne telescope. Key components include accurate PSF modelling, galaxy shape measurement, shear calibration, construction of realistic image simulations, and end-to-end pipeline validation.}
	
	\cvitem{Doctoral Research I}{Studied weak gravitational lensing of the CMB by galaxy clusters at small angular scales. Built a Maximum-a-Posteriori (MAP) estimator for cluster mass measurement targeting CMB-S4--like experiments, extending the \texttt{clusterlens} module of \chref{https://github.com/carronj/LensIt}{LensIt} (J.\ Carron). Developed flat-sky simulations of CMB patches lensed by galaxy cluster potentials.}
	
	\cvitem{Doctoral Research II}{Investigated signatures of statistical isotropy violation in the CMB arising from the motion of the observer's rest frame. Estimated the local-motion velocity at high significance using a Bayesian framework with Hamiltonian Monte Carlo sampling.}
	
	\cvitem{Master's Research}{Analysed high-energy cosmic-ray air-shower data using machine learning and deep learning techniques. Developed a Monte Carlo pipeline to simulate air showers for different primary species.}
